cmake_minimum_required(VERSION 3.24)
project(kanachan_candidate_features LANGUAGES CXX)
get_filename_component(KANACHAN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_path(KANACHAN_NYANTEN_ROOT nyanten/replacement_number.hpp HINTS ENV CPLUS_INCLUDE_PATH ENV CPATH REQUIRED)
if(NOT KANACHAN_PYTHON_INCLUDE_DIR OR NOT KANACHAN_PYTHON_LIBRARY)
  find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
  set(KANACHAN_PYTHON_INCLUDE_DIR "${Python3_INCLUDE_DIRS}")
  set(KANACHAN_PYTHON_LIBRARY "${Python3_LIBRARIES}")
endif()
add_library(_candidate_features MODULE "${KANACHAN_ROOT}/src/features/candidate_features.cpp")
target_compile_features(_candidate_features PRIVATE cxx_std_20)
target_include_directories(_candidate_features PRIVATE "${KANACHAN_NYANTEN_ROOT}" "${KANACHAN_PYTHON_INCLUDE_DIR}")
target_link_libraries(_candidate_features PRIVATE "${KANACHAN_PYTHON_LIBRARY}")
set_target_properties(_candidate_features PROPERTIES PREFIX ""
  LIBRARY_OUTPUT_DIRECTORY "${KANACHAN_ROOT}/kanachan"
  RUNTIME_OUTPUT_DIRECTORY "${KANACHAN_ROOT}/kanachan")
if(MSVC)
  target_compile_options(_candidate_features PRIVATE /utf-8 /bigobj)
  target_compile_definitions(_candidate_features PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  set_target_properties(_candidate_features PROPERTIES SUFFIX ".pyd")
endif()
